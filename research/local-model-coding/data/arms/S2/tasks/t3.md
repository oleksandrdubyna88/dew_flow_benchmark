/// <summary>A Kimi row's confinement is the shim's (D15): a plan that carries any flag is not one coai made for kimi (E5.6 decision 3).</summary>
private static void MustCarryNoFlags(ConsultantLaunch launch)
{
    if (launch.Confinement is LaunchConfinement.Planned { Plan: var plan } && (plan.Flags.Count > 0 || plan.Leading.Count > 0))

            "a Kimi row's confinement is the shim's; a plan with flags is not one coai made for kimi", nameof(launch));
    }
}
