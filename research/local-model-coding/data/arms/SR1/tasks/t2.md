```csharp
    /// <remarks>
    /// No flag, ever: Kimi's confinement is the SHIM's (D15 of todo/PLAN_vibe_and_kimi_runtimes.md), not its argv, so the plan carries none and
    /// <see cref="SandboxVocabulary"/> gains nothing. A disk row stands in the COPY of its roots (D6: a branch's AGENTS.md, skills, sub-agents and
    /// .mcp.json load from disk, M-K9) and a none row in an empty folder. Both rows are flagged unconfined by the matrix (Grep and Glob read the disk, M-K6).
    /// </remarks>
    private static Confinement.Planned Kimi(CapabilityGrant grant, AdmissionFlag flag) => grant.Capability switch
    {
        Capability.Disk => new("kimi", grant, [], [], CwdKind.Snapshot, [], flag) { SnapshotSources = grant.Roots },
        // `web` never reaches here: the matrix refuses it (no Kimi cell exercises the web).
        _ => new("kimi", grant, [], [], CwdKind.Scratch, [], flag),
    };
```
