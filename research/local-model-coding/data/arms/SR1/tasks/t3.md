```csharp
    /// <summary>Kimi's confinement is the shim's (D15), not its argv: a plan that carries a flag is not one coai made for it.</summary>
    private static void MustCarryNoFlags(ConsultantLaunch launch)
    {
        if (launch.Confinement is LaunchConfinement.Planned { Plan: { } plan } && plan.Flags.Count + plan.Leading.Count + plan.AddDirs.Count > 0)
        {
            throw new ArgumentException(
                "a Kimi row's confinement is the shim's; a plan with flags is not one coai made for kimi", nameof(launch));
        }
    }
```
